# Manual 4-player proof-of-concept run (needs Mesen Port2 = Multitap with sub-ports A-D).
# Starts from build/states/charsel_2p.mss (made from the clean ROM); P3/P4 join at
# turtle select with START on multitap pads 3/4 (Don, Raph), then are driven from them.
load build/states/charsel_2p.mss
wait 10
mtap 1 start
wait 4
mtap 1 -
wait 10
mtap 2 start
wait 4
mtap 2 -
wait 10
tap 0 b 4
wait 10
tap 1 b 4
wait 10
mtap 1 b
wait 4
mtap 1 -
wait 10
mtap 2 b
wait 4
mtap 2 -
wait 200
tap 0 start 4
wait 60
tap 0 start 4
wait 900
shot s0
peek 100 6
peek 630 32
peek 710 32
peek 1B20 48
mtap 1 right
mtap 2 right
hold 0 right
wait 60
shot s1
mtap 1 y
mtap 2 down,y
wait 60
shot s2
mtap 1 -
mtap 2 -
hold 0 -
wait 120
shot s3
peek 630 32
peek 710 32
end
